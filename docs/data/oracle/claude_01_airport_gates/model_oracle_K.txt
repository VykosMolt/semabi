domain known_vocab
type Gate(__id: str, label: str, terminal: str)
type Flight(__id: str, code: str, size: str, status: str, delay_min: int, has_assignment_flight: bool)
type Agent(__id: str, name: str, shift: str, has_assignment_agent: bool)
type Assignment(__id: str, role: str)
rel docked_at(Flight, Gate)
rel assignment_flight(Assignment, Flight)
rel assignment_agent(Assignment, Agent)
undock_flight(?f: Flight)
  pre: delay_min(?f) == '0'
  eff: docked_at(?f) := None
delay_flight(?f: Flight, ?minutes: str)
  eff: delay_min(?f) := '?minutes'
cancel_flight(?f: Flight)
  eff: status(?f) := 'cancelled'
unassign_agent(?f: Flight, ?a: Agent)
  eff: delete all x: assignment_flight(x, ?f)
dock_flight(?f: Flight, ?g: Gate)
  pre: no_incoming(docked_at, ?g) & size(?f) == 'narrow'
  eff: docked_at(?f) := ?g
assign_agent(?f: Flight, ?a: Agent, ?role: str)
  pre: no_incoming(assignment_agent, ?a)
  eff: ?new0 := new Assignment(role=?role)
  eff: assignment_flight(?new0) := ?f
  eff: assignment_agent(?new0) := ?a
