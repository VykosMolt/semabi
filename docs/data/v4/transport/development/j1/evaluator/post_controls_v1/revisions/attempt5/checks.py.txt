def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    result = importlib.util.module_from_spec(spec)
    sys.modules[name] = result
    spec.loader.exec_module(result)
    return result


def tree(public):
    nodes = []
    def add(role, name, parent, **fields):
        index = len(nodes)
        nodes.append({"i": index, "parent": parent, "role": role, "name": name,
                      "bbox": [0, 0, 10, 10], **fields})
        return index
    body = add("group", "", -1)
    add("button", "Clear message", body)
    add("button", "Reset console", body)
    for kind, label in (("transmitters", "Transmitters"), ("receivers", "Receivers"), ("patches", "Patch leads")):
        section = add("group", label, body)
        wrapper = add("group", "", section)
        for entity in public[kind]:
            group = add("group", entity["name"], wrapper)
            add("heading", entity["name"], group)
            if kind == "transmitters":
                add("button", "Transmit", group)
            elif kind == "receivers":
                label_node = add("text", "Select receiver", group)
                add("radio", "Select receiver", label_node,
                    checked=public["selected_receiver"] == entity["name"])
                add("text", "Received: " + ("yes" if entity["received"] else "no"), group)
                add("button", "Clear receipt", group)
            else:
                wrapper2 = add("group", "", group)
                for endpoint, endpoints in (("transmitter", "transmitters"), ("receiver", "receivers")):
                    label_node = add("text", endpoint.capitalize(), wrapper2)
                    add("combobox", endpoint.capitalize(), label_node,
                        value=entity[endpoint], options=[item["name"] for item in public[endpoints]])
    notice = public["notice"]
    if notice is not None:
        add("status", f"Transmission {'delivered' if notice['delivered'] else 'blocked'}: "
                      f"{notice['transmitter']} → {notice['receiver']}.", body)
    return {"url": "http://invented.invalid/join", "nodes": nodes}


def prune(raw, indices, C):
    drop = set()
    for index in indices:
        drop.update(C.descendants(raw["nodes"], index))
    remaining = [node for node in raw["nodes"] if node["i"] not in drop]
    remap = {node["i"]: index for index, node in enumerate(remaining)}
    return {"url": raw["url"], "nodes": [{**node, "i": remap[node["i"]],
             "parent": -1 if node["parent"] == -1 else remap[node["parent"]]} for node in remaining]}


def native_primitive(raw, requested, C):
    kind = requested["kind"]
    if kind in {"reset", "reload", "noop"}:
        return {"kind": kind, **({"text": requested["text"]} if "text" in requested else {})}
    scope = requested.get("scope")
    matches = [node for node in raw["nodes"] if node["role"] == requested["role"] and node["name"] == requested["name"]
               and (scope is None or any(raw["nodes"][i]["role"] == scope["role"] and raw["nodes"][i]["name"] == scope["name"]
                                        for i in C.ancestors(raw["nodes"], node["i"])[1:]))]
    assert len(matches) == 1
    node = matches[0]
    result = {"kind": kind, "target": node["i"], "target_desc": {
        "role": node["role"], "name": node["name"], "placeholder": node.get("placeholder")}}
    if kind == "select":
        result["text"] = requested["value"]
    return result


def rec(name, fields):
    return {"$record": name, "fields": fields}


def encode(value, C):
    if value is None or type(value) in (bool, int, float, str):
        return value
    if type(value) is list:
        return [encode(item, C) for item in value]
    if type(value) is tuple:
        return {"$tuple": [encode(item, C) for item in value]}
    if type(value) in (set, frozenset):
        return {"$set": type(value).__name__, "items": sorted([encode(item, C) for item in value], key=C.canonical)}
    if type(value) is dict and set(value) == {"$record", "fields"}:
        return {"$record": value["$record"], "fields": {key: encode(item, C) for key, item in value["fields"].items()}}
    if type(value) is dict:
        return {"$mapping": "dict", "items": [[encode(key, C), encode(item, C)] for key, item in value.items()]}
    raise ValueError(type(value))


def invented_values(raw, catalog, C):
    public = C.public_state(raw, catalog)
    objects, instance_list, node_instance, node_key, statics = {}, [], {}, {}, {}
    by_name = {}
    for tid, kind in enumerate(C.KINDS):
        for name in catalog[kind]:
            entity = public["entities"][name]
            root = entity["root"]["value"]
            obj = rec(C.ABSOBJ, {"tid": tid, "key": name, "attrs": {}, "parent": None,
                                "refs": {}, "ordinal": 0, "node": root, "positional": False})
            objects[(tid, name)] = obj
            by_name[name] = obj
            idx = len(instance_list)
            slots = {"id": ("", name)}
            for node_id in entity["owned_nodes"]:
                node = raw["nodes"][node_id]
                node_instance[node_id] = idx
                if node["role"] not in {"radio", "text", "combobox"}:
                    continue
                key = node["role"] + "@" + str(node_id)
                node_key[node_id] = key
                value = node.get("checked") if node["role"] == "radio" else node.get("value") if node["role"] == "combobox" else node["name"]
                slots[key] = (node["name"], value)
            instance_list.append(rec(C.INSTANCE, {"root": root, "tid": tid, "parent": None,
                                                 "slots": slots, "context": {}, "anchor": "invented", "positional": False}))
    for name in catalog["patches"]:
        by_name[name]["fields"]["refs"] = {
            "source_slot": (0, public["entities"][name]["fields"]["transmitter"]["value"]),
            "target_slot": (1, public["entities"][name]["fields"]["receiver"]["value"])}
    children = {}
    for node in raw["nodes"]:
        children.setdefault(node["parent"], []).append(node["i"])
    typed_nodes = [rec("semabi.compiler.observation.Node", {
        **{key: node.get(key) for key in ("i", "parent", "role", "name", "value", "checked", "options", "placeholder", "current")},
        "bbox": tuple(node["bbox"])}) for node in raw["nodes"]]
    obs = rec("semabi.compiler.observation.Observation", {"nodes": typed_nodes, "url": raw["url"], "_children": children})
    parsed = rec(C.PARSED, {"obs": obs, "instances": instance_list, "statics": statics,
                           "node_instance": node_instance, "node_key": node_key, "row_named": set()})
    state = rec(C.ABSSTATE, {"objs": objects, "view": {}, "partial": True, "unidentified": [],
                            "provisional": set(), "parsed": parsed, "unknown_is_none": True})
    source = catalog["transmitters"][0]
    target = public["selected"]["value"]
    event = "Transmission delivered : <> → <> ."
    vouch = rec("semabi.compiler.v4.outcome.Vouch", {
        "event": event, "witnesses": (0,), "condition": (), "covers": 1, "sole": True, "preceded_by": ()})
    return {"control": "invented-control", "state": state, "parsed": parsed, "owner": by_name[source],
            "bound": {"owner": by_name[source], "selected": by_name[target]},
            "binding_status": {"owner": "named", "selected": "named"},
            "literals": {("named", "owner")},
            "roles": {name: rec("semabi.compiler.v4.outcome.Role", {
                "name": name, "kind": kind, "form": (), "tid": tid, "anchor": None})
                for name, kind, tid in (("owner", "owner", 0), ("selected", "selection", 1))},
            "arg_roles": {event: {0: "owner", 1: "selected"}}, "decision_list": event,
            "rule": {event: vouch}, "list": {event: vouch}, "arguments": {event: {0: source, 1: target}}}


def forecast(values, C):
    return {"schema": "semabi.j1.forecast.v1", "status": "FORECAST", "instrument_status": "COMPLETE",
            "incomplete_reasons": [], "snapshots": {}, "values": encode(values, C)}
