def oracle(proj):

 def walk(v,path=(),active=()):
  if type(v) is list:
   for i,x in enumerate(v): walk(x,path+(i,),active)
  elif type(v) is dict:
   if set(v)=={'$snapshot'}:
    k=v['$snapshot']
    if k in active or k not in snaps: raise ValueError
    walk(snaps[k],path,active+(k,)); return
   if v.get('$record')==new.EVIDENCE_RECORD:
    raw=v['fields']['_blocks']; new.evidence_pair_blocks(v['fields'])
    rows.append({'path':list(path),'populated':raw is not None,'value_sha256':new.trace.sha(raw)})

    walk(x,path+(k,),active)
