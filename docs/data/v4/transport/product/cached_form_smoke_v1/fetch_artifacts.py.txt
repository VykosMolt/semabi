"""Read authorized operation artifacts through the public local HTTP API."""
import datetime
import hashlib
import json
import os
from pathlib import Path
import subprocess
import time
import urllib.request
import uuid

ROOT = Path(__file__).resolve().parents[2]
OUT = Path(__file__).resolve().parent
SPECS = {
    "memos": ("conn_0e51f4498eb547f6ac639acc96f8bab1", "op_da6cd313fd7951e26d2c", 3),
    "linkding": ("conn_69dde230fe564df8899a808d6cfd5d20", "op_5c8e47ab193b2c074b8e", 1),
}


def save(path, value):
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    with os.fdopen(fd, "w") as stream:
        json.dump(value, stream, indent=2)
        stream.write("\n")


token = (ROOT / "runs/product_service_v1/token").read_text().strip()
started = time.monotonic()
requests = []


def get(path):
    url = "http://127.0.0.1:8860" + path
    request = urllib.request.Request(url, headers={"Authorization": "Bearer " + token})
    begin = time.monotonic()
    with urllib.request.urlopen(request, timeout=20) as response:
        payload = response.read()
        requests.append({"method": "GET", "url": url, "status": response.status,
                         "response_sha256": hashlib.sha256(payload).hexdigest(),
                         "elapsed_seconds": time.monotonic() - begin})
    return json.loads(payload)


for app, (connection_id, operation_id, version) in SPECS.items():
    directory = OUT / app
    directory.mkdir(mode=0o700)
    operation = get(f"/v1/connections/{connection_id}/operations/{operation_id}?version={version}")
    connection = get(f"/v1/connections/{connection_id}")
    assert operation["id"] == operation_id and operation["version"] == version
    assert operation["status"] == "ACTIVE"
    save(directory / "operation.json", operation)
    save(directory / "connection.json", connection)
    token_part = uuid.uuid4().hex[:12]
    arguments = {}
    for name, prop in operation["argument_schema"]["properties"].items():
        if prop.get("enum"):
            value = prop["enum"][0]
        elif prop.get("type") == "boolean":
            value = False
        else:
            assert prop.get("type") == "string"
            if prop.get("format") == "uri":
                value = "https://example.invalid/baseline-" + token_part
            elif name.casefold() == "tags":
                value = "baseline" + token_part
            else:
                value = "Baseline smoke " + name.replace("_", " ") + " " + token_part
            maximum = prop.get("maxLength")
            assert maximum is None or len(value) <= maximum
            assert len(value) >= prop.get("minLength", 0)
        arguments[name] = value
    save(directory / "arguments.json", arguments)
    print(json.dumps({"app": app, "operation_id": operation_id, "version": version,
        "argument_schema": operation["argument_schema"], "entry_url": operation["procedure"]["entry_url"],
        "navigation": operation["procedure"].get("navigation", []), "arguments": arguments,
        "operation_keys": sorted(operation), "connection_keys": sorted(connection)}))
token = None
