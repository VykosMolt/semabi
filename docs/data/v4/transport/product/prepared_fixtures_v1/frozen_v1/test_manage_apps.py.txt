def _change_data(app):
    (app / "data/database.sqlite").write_bytes(b"later opaque database state")
    (app / "data/database.sqlite-wal").write_bytes(b"later opaque WAL state")
    (app / "data/attachments/picture.bin").unlink()
    (app / "data/new-directory").mkdir(mode=0o700)
    (app / "data/new-directory/new-file").write_bytes(b"later attachment")


def _saved(app, name="base"):
    return app / "private/prepared" / name


def _manifest_mutation(app, mutate):
    path = _saved(app) / "manifest.json"
    value = json.loads(path.read_text())
    mutate(value)
    path.write_text(json.dumps(value))


@pytest.mark.parametrize("was_running", [True, False])
def test_exact_opaque_restore_preserves_modes_credentials_and_archive(installation, was_running):
    app, manifest, state = installation
    state["running"] = was_running
    before = helper._inventory(app / "data")
    external = {p: (p.read_bytes(), stat.S_IMODE(p.stat().st_mode))
                for p in [app / "private/credentials.json", app / "server.env"]}
    saved = helper.snapshot(app, manifest, "base")
    assert saved["was_running"] is was_running
    assert state["calls"] == (["stop", "start"] if was_running else [])
    assert helper._inventory(_saved(app) / "data") == before
    saved_bytes = (_saved(app) / "manifest.json").read_bytes()
    for path in [app / "private", app / "private/prepared", _saved(app)]:
        assert stat.S_IMODE(path.stat().st_mode) == 0o700
    assert stat.S_IMODE((_saved(app) / "manifest.json").stat().st_mode) == 0o600
    _change_data(app)
    changed = helper._inventory(app / "data")
    state["calls"].clear()
    restored = helper.restore(app, manifest, "base")
    assert restored["was_running"] is was_running
    assert state["running"] is was_running
    assert state["calls"] == (["stop", "start"] if was_running else [])
    assert helper._inventory(app / "data") == before
    assert helper._inventory(Path(restored["archive"])) == changed
    assert helper._inventory(_saved(app) / "data") == before
    assert (_saved(app) / "manifest.json").read_bytes() == saved_bytes
    assert restored["manifest_sha256"] == saved["manifest_sha256"]
    assert all((p.read_bytes(), stat.S_IMODE(p.stat().st_mode)) == values for p, values in external.items())
    assert stat.S_IMODE((app / "private/resets").stat().st_mode) == 0o700
    assert not list((app / "private").glob(".restore-*"))

@pytest.mark.parametrize("damage", [
    "content", "missing_file", "extra_file", "missing_empty_directory", "mode",
    "application", "version", "deployment", "traversal", "absolute", "long_path",
    "missing_parent", "duplicate_path", "duplicate_manifest_key", "bad_manifest_mode",
    "manifest_symlink", "manifest_fifo", "data_symlink", "data_fifo", "extra_container_entry",
])
def test_invalid_prepared_fixture_is_rejected_before_stop_or_staging(installation, damage):
    app, manifest, state = installation
    helper.snapshot(app, manifest, "base")
    snapshot = _saved(app)
    data_file = snapshot / "data/database.sqlite"
    if damage == "content":
        data_file.write_bytes(b"tampered")
    elif damage == "missing_file":
        data_file.unlink()
    elif damage == "extra_file":
        (snapshot / "data/unlisted").write_bytes(b"extra")
    elif damage == "missing_empty_directory":
        (snapshot / "data/attachments/empty").rmdir()
    elif damage == "mode":
        data_file.chmod(0o640)
    elif damage in {"application", "version", "deployment"}:
        manifest[{"application": "application", "version": "version", "deployment": "port"}[damage]] = "different"
    elif damage in {"traversal", "absolute", "long_path", "missing_parent"}:
        path = {"traversal": "../escape", "absolute": "/escape", "long_path": "a" * 4097,
                "missing_parent": "absent-parent/file"}[damage]
        _manifest_mutation(app, lambda value: value["inventory"][1].update(path=path))
    elif damage == "duplicate_path":
        _manifest_mutation(app, lambda value: value["inventory"].append(value["inventory"][-1]))
    elif damage == "duplicate_manifest_key":
        path = snapshot / "manifest.json"
        path.write_text(path.read_text().replace('"format":', '"format":"duplicate","format":', 1))
    elif damage == "bad_manifest_mode":
        (snapshot / "manifest.json").chmod(0o644)
    elif damage in {"manifest_symlink", "manifest_fifo"}:
        path = snapshot / "manifest.json"
        path.unlink()
        if damage == "manifest_symlink":
            path.symlink_to(app / "private/credentials.json")
        else:
            os.mkfifo(path, 0o600)
    elif damage in {"data_symlink", "data_fifo"}:
        data_file.unlink()
        if damage == "data_symlink":
            data_file.symlink_to(app / "private/credentials.json")
        else:
            os.mkfifo(data_file, 0o600)
    elif damage == "extra_container_entry":
        (snapshot / "unexpected").write_text("unexpected")
    before = helper._inventory(app / "data")
    private_names = sorted(p.name for p in (app / "private").iterdir())
    state["calls"].clear()
    with pytest.raises((ValueError, OSError)):
        helper.restore(app, manifest, "base")
    assert state["calls"] == []
    assert state["running"] is True
    assert helper._inventory(app / "data") == before
    assert sorted(p.name for p in (app / "private").iterdir()) == private_names

@pytest.mark.parametrize("failure", ["archive", "install", "rollback", "destination_appears"])
def test_restore_rename_failures_keep_prior_data_recoverable(installation, monkeypatch, failure):
    app, manifest, state = installation
    helper.snapshot(app, manifest, "base")
    snapshot_inventory = helper._inventory(_saved(app) / "data")
    _change_data(app)
    before = helper._inventory(app / "data")
    state["calls"].clear()
    original = helper._rename_no_replace

    def fail_rename(source, destination):
        if failure == "archive" and source == app / "data":
            raise OSError("synthetic archive failure")
        if destination == app / "data" and source.parent.name.startswith(".restore-"):
            if failure == "destination_appears":
                destination.mkdir(mode=0o700)
                (destination / "preserve-existing").write_bytes(b"do not replace")
            raise OSError("synthetic install failure")
        if failure == "rollback" and destination == app / "data":
            raise OSError("synthetic rollback failure")
        return original(source, destination)

    monkeypatch.setattr(helper, "_rename_no_replace", fail_rename)
    with pytest.raises(OSError, match="synthetic") as caught:
        helper.restore(app, manifest, "base")
    archives = list((app / "private/resets").iterdir())
    if failure in {"archive", "install"}:
        assert helper._inventory(app / "data") == before
        assert archives == []
        assert state["calls"] == ["stop", "start"]
    else:
        assert len(archives) == 1
        assert helper._inventory(archives[0]) == before
        if failure == "rollback":
            assert not (app / "data").exists()
            assert state["calls"] == ["stop"]
            assert state["running"] is False
            assert any("Rollback also failed" in note for note in caught.value.__notes__)
        else:
            assert (app / "data/preserve-existing").read_bytes() == b"do not replace"
    assert helper._inventory(_saved(app) / "data") == snapshot_inventory
    assert not list((app / "private").glob(".restore-*"))

def test_app_directory_lock_serializes_mutations_and_allows_nested_flow(tmp_path):
    app = tmp_path / "memos"
    app.mkdir()
    attempted = threading.Event()
    completed = threading.Event()

    @helper._serialized
    def mutation(app, manifest):
        with helper._lifecycle_lock(app):
            completed.set()

    def worker():
        attempted.set()
        mutation(app, {})

    with helper._lifecycle_lock(app):
        thread = threading.Thread(target=worker, daemon=True)
        thread.start()
        assert attempted.wait(1)
        assert not completed.wait(0.05)
    thread.join(2)
    assert not thread.is_alive() and completed.is_set()
    assert list(app.iterdir()) == []


@pytest.mark.parametrize("action", ["snapshot", "restore"])
def test_cli_accepts_application_and_name_without_real_lifecycle_calls(installation, monkeypatch, capsys, action):
    app, manifest, state = installation
    monkeypatch.setattr(helper, "ROOT", app.parent)
    monkeypatch.setattr(helper, "status", lambda app, manifest: {"application": "memos", "running": False})
    calls = []

    def command(app, manifest, name):
        calls.append((app, name))
        return {"action": action, "name": name}

    monkeypatch.setattr(helper, action, command)
    monkeypatch.setattr(helper.sys, "argv", ["manage_apps.py", action, "memos", "base-1"])
    helper.main()
    assert calls == [(app, "base-1")]
    assert json.loads(capsys.readouterr().out)["prepared_fixture"] == {"action": action, "name": "base-1"}
    assert state["calls"] == []
